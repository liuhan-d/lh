一、条件语句 if / else if / else（成绩评等级）
if (lh_score >= 90){
    lh_grade = "优秀";
} else if (lh_score >= 80){
    lh_grade = "良好";
} else if (lh_score >= 60){
    lh_grade = "及格";
} else {
    lh_grade = "不及格";
}

二、条件语句 switch（匹配星期几）
switch (lh_day){
    case 1: lh_name = "星期一"; break;
    case 2: lh_name = "星期二"; break;
    case 3: lh_name = "星期三"; break;
    ...
    default: lh_name = "输入有误";
}
三、循环语句 for（1~100 累加求和）
var lh_sum = 0;
for (var lh_i = 1; lh_i <= 100; lh_i++){
    lh_sum = lh_sum + lh_i;   // 循环体执行 100 次
}
四、循环语句 while（求 5 的阶乘）
var lh_n = 5, lh_fact = 1;
while (lh_n > 0){          // 条件为 true 就一直循环
    lh_fact = lh_fact * lh_n;
    lh_n--;                // 别忘了改变条件，避免死循环
}
五、综合运用：双重 for 循环（九九乘法表）
for (var lh_i = 1; lh_i <= 9; lh_i++){        // 外层循环：控制行
    for (var lh_j = 1; lh_j <= lh_i; lh_j++){ // 内层循环：控制列
        // 每行生成 lh_j × lh_i = 积
    }
}
