JavaScript 对象

刘晗的《高级网页设计》课堂实践 · 创建对象、属性访问、方法与 this、for...in 遍历、构造函数

一、创建对象的三种方式
// 方式 1：对象字面量（最常用）
var lh_student = {
    lh_name: "刘晗",
    lh_age: 20,
    lh_major: "网页设计"
};

// 方式 2：new Object()
var lh_car = new Object();
lh_car.lh_brand = "红旗";
lh_car.lh_price = 200000;

// 方式 3：构造函数（批量创建同类对象）
function lh_Person(lh_name, lh_age){
    this.lh_name = lh_name;
    this.lh_age = lh_age;
}
var lh_p = new lh_Person("小明", 19);
二、访问属性：点语法 与 方括号语法
var lh_val1 = lh_student.lh_name;      // 点语法：对象.属性名
var lh_key = "lh_major";
var lh_val2 = lh_student[lh_key];      // 方括号：属性名是变量时必须用它

属性名保存在变量里（或含空格等特殊字符）时，只能用方括号语法：

点击按钮查看 lh_student 的属性值。
三、对象的方法 与 this 关键字
var lh_student = {
    lh_name: "刘晗",
    lh_age: 20,
    lh_sayHi: function(){            // 方法：属性值是函数
        return "大家好，我是 " + this.lh_name;  // this 指当前对象
    },
    lh_addAge: function(){
        this.lh_age = this.lh_age + 1;          // 通过 this 修改自身属性
    }
};
点击按钮测试对象的方法。
四、遍历对象：for...in 循环 与 Object.keys()
// for...in：依次取出每一个属性名
for (var lh_key in lh_student){
    // lh_key 是属性名，lh_student[lh_key] 是属性值
}

// Object.keys()：把所有属性名收集成数组
var lh_keys = Object.keys(lh_student);
五、动态添加与删除属性
lh_student.lh_score = 95;          // 添加新属性：直接赋值
delete lh_student.lh_score;        // 删除属性：delete 运算符
点击按钮动态增删属性。
六、综合运用：构造函数批量创建学生对象
function lh_Student(lh_name, lh_age, lh_score){
    this.lh_name = lh_name;
    this.lh_age = lh_age;
    this.lh_score = lh_score;
    this.lh_intro = function(){
        return this.lh_name + "（" + this.lh_age + " 岁，" + this.lh_score + " 分）";
    };
}
var lh_arr = [
    new lh_Student("刘晗", 20, 95),
    new lh_Student("小明", 19, 88),
    new lh_Student("小红", 21, 92)
];
